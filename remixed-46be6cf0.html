UPDATE и DELETE: изменение данных
✓
▶
-- UPDATE: обновление UPDATE users SET age = 22 WHERE id = 1; UPDATE users SET city = 'Астана', is_active = false WHERE email = 'ali@example.com'; -- UPDATE с вычислением UPDATE products SET price = price * 1.1 WHERE category = 'Телефоны'; -- повысить цену на 10% -- RETURNING — что обновилось UPDATE users SET age = 25 WHERE id = 3 RETURNING id, name, age; -- DELETE: удаление DELETE FROM users WHERE id = 10; DELETE FROM users WHERE is_active = false; -- UPSERT: вставить или обновить INSERT INTO users (email, name, age) VALUES ('ali@test.com', 'Алишер', 22) ON CONFLICT (email) DO UPDATE SET age = EXCLUDED.age; -- обновить если email уже есть
⚠ ОПАСНО

DELETE FROM users; без WHERE удалит ВСЕ записи! UPDATE users SET is_active = false; без WHERE деактивирует всех! Всегда проверяй: сначала напиши SELECT с тем же WHERE, убедись что нашёл нужные записи — потом UPDATE/DELETE.

07
ORDER BY, LIMIT, OFFSET и пагинация
✓
▶

Сортировка и пагинация — основа любого списка в приложении. ORDER BY задаёт порядок вывода, LIMIT ограничивает количество строк, OFFSET пропускает строки — именно так работает пагинация (страница 1, 2, 3...). Понимание этого критично для любого backend API.

-- Сортировка SELECT * FROM products ORDER BY price; -- по возрастанию (ASC по умолчанию) SELECT * FROM products ORDER BY price DESC; -- по убыванию SELECT * FROM users ORDER BY city, name; -- сначала по городу, затем по имени SELECT * FROM orders ORDER BY created_at DESC; -- последние заказы первыми -- NULL в сортировке ORDER BY price NULLS LAST -- NULL-значения в конце ORDER BY price NULLS FIRST -- NULL-значения первыми -- Ограничение результатов SELECT * FROM products ORDER BY price DESC LIMIT 10; -- топ-10 дорогих товаров SELECT * FROM users LIMIT 1; -- первая запись -- Пагинация: LIMIT + OFFSET -- Страница 1 (записи 1-10) SELECT * FROM products ORDER BY id LIMIT 10 OFFSET 0; -- Страница 2 (записи 11-20) SELECT * FROM products ORDER BY id LIMIT 10 OFFSET 10; -- Страница N -- OFFSET = (page_number - 1) * page_size -- Формула пагинации из Python (FastAPI/Django) -- page=3, size=10 → LIMIT 10 OFFSET 20 -- cursor.execute("SELECT * FROM products ORDER BY id LIMIT %s OFFSET %s", (size, (page-1)*size)) -- Считаем общее количество страниц SELECT COUNT(*) FROM products; -- total / page_size = кол-во страниц
⚠ Проблема больших OFFSET

При большом OFFSET (например 100000) PostgreSQL всё равно читает все строки до нужного места — это медленно. Для больших таблиц используй cursor-based pagination: WHERE id > last_seen_id LIMIT 10.

08
Первичный и внешний ключи: PRIMARY KEY, FOREIGN KEY
✓
▶

PRIMARY KEY — уникальный идентификатор строки. FOREIGN KEY — ссылка на PK другой таблицы. Обеспечивает ссылочную целостность: нельзя добавить заказ несуществующему пользователю.

Связь «один ко многим»: users orders ┌─────────────────┐ ┌─────────────────────────┐ │ id (PK) │◄──────────│ id (PK) │ │ name │ 1 : MANY │ user_id (FK → users.id) │ │ email │ │ total_price │ └─────────────────┘ │ status │ └─────────────────────────┘ Один пользователь может иметь много заказов.
-- Создание таблицы с FK CREATE TABLE orders ( id SERIAL PRIMARY KEY, user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE, -- удалить заказы при удалении юзера total NUMERIC(10,2) NOT NULL, status VARCHAR(20) DEFAULT 'pending', created_at TIMESTAMP DEFAULT NOW() ); -- Добавить заказы INSERT INTO orders (user_id, total, status) VALUES (1, 1299.99, 'completed'), (1, 249.99, 'pending'), (2, 999.99, 'completed'), (3, 49.99, 'cancelled'); -- Это вызовет ошибку (пользователя 999 нет): -- INSERT INTO orders (user_id, total) VALUES (999, 100); -- ERROR: violates foreign key constraint

09
INNER JOIN: объединение таблиц
✓
▶

JOIN объединяет строки из двух таблиц по условию. INNER JOIN возвращает только строки у которых есть совпадения в обеих таблицах.

INNER JOIN: users: [1:Алишер] [2:Бекзат] [3:Диана] [4:Нурлан] orders: [u_id=1] [u_id=1] [u_id=2] [u_id=5 ← нет в users!] INNER JOIN вернёт только: Алишер+его заказы, Бекзат+его заказы Нурлан (без заказов) и u_id=5 (нет пользователя) — пропущены
-- Базовый INNER JOIN SELECT u.name, u.email, o.id AS order_id, o.total, o.status FROM users u INNER JOIN orders o ON u.id = o.user_id; -- JOIN с фильтрацией SELECT u.name, o.total FROM users u JOIN orders o ON u.id = o.user_id WHERE o.status = 'completed' ORDER BY o.total DESC; -- Тройной JOIN SELECT u.name AS customer, p.name AS product, oi.quantity, oi.price FROM users u JOIN orders o ON u.id = o.user_id JOIN order_items oi ON o.id = oi.order_id JOIN products p ON oi.product_id = p.id;

10
LEFT JOIN, RIGHT JOIN, FULL JOIN
✓
▶
Типы JOIN визуально: INNER JOIN: [A ∩ B] только совпадения LEFT JOIN: [A] все левые + совпадения справа (NULL если нет) RIGHT JOIN: [B] все правые + совпадения слева (NULL если нет) FULL JOIN: [A ∪ B] все из обеих таблиц Пример LEFT JOIN: users: Алишер(2 заказа) Бекзат(1 заказ) Диана(0 заказов) LEFT JOIN: Алишер+заказы Бекзат+заказ Диана + NULL NULL
-- LEFT JOIN: все пользователи, даже без заказов SELECT u.name, o.id AS order_id, o.total, COALESCE(o.total, 0) AS total_or_zero -- NULL → 0 FROM users u LEFT JOIN orders o ON u.id = o.user_id; -- Найти пользователей БЕЗ заказов SELECT u.name, u.email FROM users u LEFT JOIN orders o ON u.id = o.user_id WHERE o.id IS NULL; -- нет совпадения = нет заказов -- FULL OUTER JOIN SELECT u.name, o.total FROM users u FULL OUTER JOIN orders o ON u.id = o.user_id;

11
SELF JOIN и алиасы таблиц
✓
▶

SELF JOIN — соединение таблицы с самой собой. Используется для иерархических данных: сотрудник → руководитель, комментарий → родительский комментарий, категория → родительская категория. Алиасы (AS) делают запросы читаемыми и позволяют различать копии одной таблицы.

-- Таблица сотрудников с иерархией CREATE TABLE employees ( id SERIAL PRIMARY KEY, name VARCHAR(100), manager_id INT REFERENCES employees(id) -- ссылка на себя! ); -- SELF JOIN: сотрудники со своими руководителями SELECT e.name AS employee, m.name AS manager FROM employees e LEFT JOIN employees m ON e.manager_id = m.id; -- LEFT JOIN чтобы показать и тех у кого нет руководителя (NULL) -- Алиасы для удобства и читаемости SELECT u.name AS user_name, p.title AS product_name, o.total AS order_total, o.created_at AS ordered_at FROM orders o JOIN users u ON o.user_id = u.id JOIN products p ON o.product_id = p.id; -- Категории с подкатегориями (рекурсивная структура) SELECT child.name AS subcategory, parent.name AS category FROM categories child LEFT JOIN categories parent ON child.parent_id = parent.id;

12
UNION, INTERSECT, EXCEPT: объединение результатов
✓
▶

Операторы множеств комбинируют результаты нескольких SELECT. UNION объединяет (убирая дубликаты), UNION ALL — с дубликатами, INTERSECT — только общие строки, EXCEPT — строки первого запроса которых нет во втором.

-- UNION: объединить клиентов из двух городов SELECT name, email FROM users WHERE city = 'Алматы' UNION SELECT name, email FROM users WHERE city = 'Астана'; -- Колонок должно быть одинаково, типы совместимы -- UNION ALL (быстрее — не убирает дубликаты) SELECT product_id, 'active' AS status FROM active_products UNION ALL SELECT product_id, 'archived' AS status FROM archived_products; -- INTERSECT: пользователи которые есть И там и там SELECT user_id FROM orders WHERE status = 'completed' INTERSECT SELECT user_id FROM reviews; -- Пользователи которые и купили И оставили отзыв -- EXCEPT: пользователи которые купили но НЕ оставили отзыв SELECT user_id FROM orders WHERE status = 'completed' EXCEPT SELECT user_id FROM reviews; -- Сортировка применяется к итоговому результату SELECT name FROM admins UNION SELECT name FROM moderators ORDER BY name;
Задача дня

Используя UNION выведи единый список всех пользователей из Алматы и Астаны. Через EXCEPT найди пользователей у которых ЕСТЬ аккаунт, но НЕТ ни одного заказа (подсказка: SELECT id FROM users EXCEPT SELECT user_id FROM orders). Объясни разницу между UNION и UNION ALL.

13
WITH (CTE): общие табличные выражения
✓
▶

CTE (Common Table Expression) — временный именованный результат запроса. Делает сложные запросы читаемыми: вместо вложенных подзапросов ты описываешь логику шаг за шагом. Можно использовать несколько CTE в одном запросе.

-- Базовый синтаксис WITH cte_name AS ( SELECT ... -- этот результат доступен ниже ) SELECT * FROM cte_name; -- Пример: топ покупатели с их последним заказом WITH top_buyers AS ( SELECT user_id, SUM(total) AS spent FROM orders GROUP BY user_id HAVING SUM(total) > 10000 ), last_orders AS ( SELECT DISTINCT ON (user_id) user_id, created_at AS last_order_date FROM orders ORDER BY user_id, created_at DESC ) SELECT u.name, tb.spent, lo.last_order_date FROM top_buyers tb JOIN users u ON tb.user_id = u.id JOIN last_orders lo ON tb.user_id = lo.user_id ORDER BY tb.spent DESC; -- CTE делает код читаемым — каждый блок описывает один шаг логики
💡 CTE vs Подзапрос

CTE проще читать: каждый шаг имеет имя и понятную роль. Подзапрос в FROM (derived table) работает так же, но CTE можно переиспользовать в одном запросе несколько раз. При сложной логике — всегда CTE.

Задача дня

14
Многие-ко-многим: связующие таблицы
✓
▶

Связь «многие-ко-многим» (M:N) — самая частая в реальных приложениях. Товар может иметь много тегов, тег — много товаров. Статья — много авторов, автор — много статей. Реализуется через промежуточную (связующую) таблицу.

-- Пример: статьи и теги (M:N) CREATE TABLE posts ( id SERIAL PRIMARY KEY, title VARCHAR(200) ); CREATE TABLE tags ( id SERIAL PRIMARY KEY, name VARCHAR(50) UNIQUE ); -- Связующая таблица CREATE TABLE post_tags ( post_id INT REFERENCES posts(id) ON DELETE CASCADE, tag_id INT REFERENCES tags(id) ON DELETE CASCADE, PRIMARY KEY (post_id, tag_id) -- составной PK — дублей нет ); -- Добавить тег к статье INSERT INTO post_tags (post_id, tag_id) VALUES (1, 3); -- Все статьи с тегами SELECT p.title, STRING_AGG(t.name, ', ') AS tags FROM posts p JOIN post_tags pt ON p.id = pt.post_id JOIN tags t ON pt.tag_id = t.id GROUP BY p.id, p.title; -- Все статьи с конкретным тегом SELECT p.title FROM posts p JOIN post_tags pt ON p.id = pt.post_id JOIN tags t ON pt.tag_id = t.id WHERE t.name = 'python';

Индексы: ускорение поиска
✓
▶

Индекс — дополнительная структура данных для быстрого поиска. Как оглавление в книге: вместо просмотра всех страниц — сразу переходишь нужную. Замедляет INSERT/UPDATE (нужно обновлять индекс), но ускоряет SELECT в тысячи раз.

-- Создание индексов CREATE INDEX idx_users_email ON users(email); CREATE INDEX idx_users_city ON users(city); CREATE INDEX idx_orders_user_id ON orders(user_id); -- для JOIN CREATE INDEX idx_orders_status_date ON orders(status, created_at); -- составной -- Уникальный индекс (автоматически создаётся для UNIQUE/PK) CREATE UNIQUE INDEX idx_users_email_unique ON users(email); -- Посмотреть план запроса (есть ли использование индекса) EXPLAIN SELECT * FROM users WHERE email = 'ali@test.com'; -- Seq Scan = полный перебор (плохо) -- Index Scan = использует индекс (хорошо) EXPLAIN ANALYZE SELECT * FROM users WHERE city = 'Алматы'; -- ANALYZE реально выполняет запрос и показывает время -- Посмотреть все индексы таблицы \d users -- или SELECT * FROM pg_indexes WHERE tablename = 'users';
⚠ Когда создавать индексы

Создавай индексы на: поля в WHERE которые используешь часто, поля JOIN (Foreign Keys), поля ORDER BY в больших таблицах. Не создавай на маленьких таблицах (< 1000 строк) — оптимизатор всё равно выберет Seq Scan.

Оконные функции: ROW_NUMBER, RANK, LAG, LEAD
✓
▶

Оконные функции вычисляют значения по группе строк (окну) не схлопывая их в одну — в отличие от GROUP BY. Это мощнейший инструмент аналитики: рейтинги, нарастающие итоги, сравнение с предыдущим периодом.

-- Синтаксис: функция OVER (PARTITION BY ... ORDER BY ...) -- ROW_NUMBER: порядковый номер в группе SELECT name, city, ROW_NUMBER() OVER (PARTITION BY city ORDER BY name) AS row_in_city FROM users; -- RANK: рейтинг (с пропуском при одинаковых) -- DENSE_RANK: рейтинг (без пропуска) SELECT name, total, RANK() OVER (ORDER BY total DESC) AS rank FROM orders; -- LAG/LEAD: предыдущее/следующее значение SELECT month, revenue, LAG(revenue) OVER (ORDER BY month) AS prev_month, revenue - LAG(revenue) OVER (ORDER BY month) AS growth FROM monthly_stats; -- Нарастающий итог (running total) SELECT created_at, total, SUM(total) OVER (ORDER BY created_at) AS cumulative_sum FROM orders; -- Топ-1 заказ каждого пользователя SELECT * FROM ( SELECT *, ROW_NUMBER() OVER (PARTITION BY user_id ORDER BY total DESC) AS rn FROM orders ) sub WHERE rn = 1;
💡 Зачем это нужно

Оконные функции — обязательный навык для аналитических запросов. Используются везде: рейтинг пользователей, отчёты по периодам, дедупликация, поиск последней записи для каждой группы. Без них нужны громоздкие подзапросы.

SQL-инъекции: атака и защита
✓
▶

SQL-инъекция — подстановка SQL-кода через поля ввода. Входит в топ-3 уязвимостей OWASP. Позволяет взломщику: получить все данные, обойти авторизацию, удалить таблицы.

-- УЯЗВИМЫЙ КОД (Python + f-строка — никогда так не делай!) username = input() # пользователь вводит: admin' OR '1'='1 query = f"SELECT * FROM users WHERE email='{username}'" # Итоговый запрос: # SELECT * FROM users WHERE email='admin' OR '1'='1' # '1'='1' всегда true → вернёт ВСЕХ пользователей! -- Ещё опаснее: DROP TABLE # Ввод: '; DROP TABLE users; -- # Итог: SELECT * FROM users WHERE email=''; DROP TABLE users; --' # Всё удалено! -- ЗАЩИТА: Prepared Statements (параметризованные запросы) import psycopg2 # БЕЗОПАСНО — данные передаются отдельно от запроса cursor.execute( "SELECT * FROM users WHERE email = %s", (username,) # кортеж с данными ) # psycopg2 экранирует спецсимволы автоматически # ' становится \' — атака не работает -- Также защищает использование ORM: # User.query.filter_by(email=username).first() # Django/SQLAlchemy

EXPLAIN ANALYZE: план выполнения запроса
✓
▶

EXPLAIN ANALYZE — рентген запроса. Показывает как PostgreSQL выполняет запрос: какой план выбрал оптимизатор, сколько строк обработал, сколько времени ушло на каждый шаг. Без этого невозможно оптимизировать медленные запросы.

-- EXPLAIN — только план (без выполнения) EXPLAIN SELECT * FROM users WHERE email = 'test@test.com'; -- EXPLAIN ANALYZE — план + реальное выполнение EXPLAIN ANALYZE SELECT * FROM users WHERE email = 'test@test.com'; -- Что читать в выводе: -- Seq Scan → полное сканирование таблицы (плохо для больших таблиц) -- Index Scan → использует индекс (хорошо) -- Bitmap Heap Scan → гибрид -- rows= → оценка/реальное количество строк -- cost= → относительная стоимость -- actual time= → реальное время в мс -- Сравниваем ДО создания индекса: EXPLAIN ANALYZE SELECT * FROM orders WHERE status = 'pending'; -- Seq Scan on orders (cost=0.00..450.00 rows=12 width=40) -- actual time=0.023..5.234 ms CREATE INDEX idx_orders_status ON orders(status); -- ПОСЛЕ создания индекса: EXPLAIN ANALYZE SELECT * FROM orders WHERE status = 'pending'; -- Index Scan using idx_orders_status (cost=0.28..8.30 rows=12) -- actual time=0.021..0.089 ms ← намного быстрее! -- EXPLAIN ANALYZE с JOIN EXPLAIN ANALYZE SELECT u.name, COUNT(o.id) FROM users u LEFT JOIN orders o ON u.id = o.user_id GROUP BY u.id;
💡 Онлайн-визуализация

Скопируй вывод EXPLAIN в explain.dalibo.com — получишь красивую визуализацию плана запроса с деревом операций. Незаменимо для анализа сложных запросов.

Транзакции: COMMIT и ROLLBACK
✓
▶

Транзакция — набор операций которые выполняются атомарно: либо все успешно, либо ни одной. Классический пример: перевод денег — снять со счёта A и добавить на счёт B должны произойти вместе или не произойти вовсе.

-- SQL-транзакции BEGIN; UPDATE accounts SET balance = balance - 1000 WHERE id = 1; UPDATE accounts SET balance = balance + 1000 WHERE id = 2; COMMIT; -- подтвердить -- или ROLLBACK; — отменить если ошибка # Python: транзакции через psycopg2 def transfer_money(from_id, to_id, amount): conn = psycopg2.connect(DB_URL) try: with conn.cursor() as cur: # Проверить баланс cur.execute("SELECT balance FROM accounts WHERE id=%s", (from_id,)) balance = cur.fetchone()[0] if balance < amount: raise ValueError("Недостаточно средств") # Снять cur.execute( "UPDATE accounts SET balance = balance - %s WHERE id = %s", (amount, from_id) ) # Добавить cur.execute( "UPDATE accounts SET balance = balance + %s WHERE id = %s", (amount, to_id) ) conn.commit() # подтвердить обе операции print("Перевод выполнен") except Exception as e: conn.rollback() # отменить всё при ошибке print(f"Ошибка, откат: {e}") finally: conn.close()

Connection Pool: эффективное управление соединениями
✓
▶

Открытие нового соединения с PostgreSQL — дорогая операция (50-100ms). В реальном приложении на каждый HTTP-запрос создавать новое соединение — катастрофа. Connection pool держит пул готовых соединений и переиспользует их. Это стандарт для любого production-приложения.

# Без пула — медленно и ненадёжно # conn = psycopg2.connect(...) # каждый раз новое соединение — НЕ ТАК # pip install psycopg2-binary from psycopg2 import pool # Создаём пул при старте приложения connection_pool = pool.ThreadedConnectionPool( minconn=2, # минимум соединений maxconn=10, # максимум соединений host="localhost", dbname="academy", user="dev", password="secret" ) # Использование пула def get_users(): conn = connection_pool.getconn() # взять соединение из пула try: with conn.cursor() as cur: cur.execute("SELECT id, name FROM users") return cur.fetchall() finally: connection_pool.putconn(conn) # вернуть в пул! # Контекстный менеджер (чище) from contextlib import contextmanager @contextmanager def get_db(): conn = connection_pool.getconn() try: yield conn finally: connection_pool.putconn(conn) # Использование with get_db() as conn: with conn.cursor() as cur: cur.execute("SELECT COUNT(*) FROM orders") count = cur.fetchone()[0]
💡 В реальных фреймворках

Flask + SQLAlchemy, FastAPI + databases, Django — все используют connection pool под капотом. Но знать как это работает важно: настройка pool_size, max_overflow, pool_timeout напрямую влияет на производительность под нагрузкой.

Миграции базы данных: Alembic
✓
▶

Миграции — версионный контроль схемы базы данных. Когда нужно добавить колонку, переименовать таблицу или создать индекс в продакшн — нельзя просто запустить CREATE TABLE. Alembic фиксирует каждое изменение схемы и позволяет откатить его.

# pip install alembic # Инициализация alembic init alembic # создаёт папку alembic/ и alembic.ini # В alembic.ini прописать подключение: # sqlalchemy.url = postgresql://dev:secret@localhost/academy # Создать миграцию автоматически (сравнивает модели с БД) alembic revision --autogenerate -m "add phone column to users" # Применить миграцию alembic upgrade head # применить все alembic upgrade +1 # на одну вперёд # Откатить alembic downgrade -1 # на одну назад alembic downgrade base # до самого начала # История миграций alembic history alembic current # текущая версия # Пример файла миграции (versions/abc123_add_phone.py) def upgrade(): op.add_column('users', sa.Column('phone', sa.String(20))) def downgrade(): op.drop_column('users', 'phone')
🔒 Правило продакшна

Никогда не меняй схему БД напрямую в продакшн через psql. Только через миграции. Иначе локальная версия расходится с продакшном и коллеги не смогут воспроизвести проблему. Миграции — обязательная практика в командной разработке.

Задача дня

Резервное копирование PostgreSQL: pg_dump
✓
▶

Бэкапы — обязательная часть эксплуатации БД. pg_dump создаёт снимок базы данных. Без регулярных бэкапов потеря данных из-за ошибки или атаки — катастрофа без выхода. Автоматизируй бэкапы через cron.

# Бэкап одной базы (SQL-формат) $ pg_dump -U dev academy > backup_$(date +%Y%m%d).sql # Бэкап в сжатый бинарный формат (быстрее восстанавливается) $ pg_dump -U dev -Fc academy > backup_$(date +%Y%m%d).dump # Бэкап только структуры (без данных) $ pg_dump -U dev --schema-only academy > schema.sql # Бэкап только данных $ pg_dump -U dev --data-only academy > data.sql # Бэкап конкретной таблицы $ pg_dump -U dev -t users academy > users_backup.sql # Восстановление из SQL $ psql -U dev -d academy_restore < backup_20241201.sql # Восстановление из бинарного $ pg_restore -U dev -d academy_restore backup_20241201.dump # Автоматический бэкап через cron (каждый день в 3:00) # 0 3 * * * pg_dump -U dev academy > /backups/academy_$(date +\%Y\%m\%d).sql # pg_dumpall — бэкап ВСЕХ баз и пользователей $ pg_dumpall -U postgres > all_databases.sql
⚠ Проверяй бэкапы!

Бэкап который никто не проверял — не бэкап. Раз в месяц разворачивай бэкап на тестовой БД и убеждайся что данные восстанавливаются корректно. Иначе в момент аварии окажется что файл битый.
